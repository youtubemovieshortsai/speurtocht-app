function formatTime(totalSeconds: number) {
  const safeSeconds = Math.max(0, Math.floor(totalSeconds));

  const hours = Math.floor(safeSeconds / 3600);
  const minutes = Math.floor((safeSeconds % 3600) / 60);
  const seconds = safeSeconds % 60;

  return [
    hours.toString().padStart(2, "0"),
    minutes.toString().padStart(2, "0"),
    seconds.toString().padStart(2, "0"),
  ].join(":");
}

function formatSeekerTime(totalSeconds: number) {
  const safeSeconds = Math.max(0, Math.floor(totalSeconds));

  if (safeSeconds < 60) {
    return `${safeSeconds} sec`;
  }

  const minutes = Math.floor(safeSeconds / 60);
  const seconds = safeSeconds % 60;

  if (minutes < 60) {
    return `${minutes} min ${seconds} sec`;
  }

  const hours = Math.floor(minutes / 60);
  const remainingMinutes = minutes % 60;

  return `${hours} uur ${remainingMinutes} min`;
}

"use client";

import { MapContainer, TileLayer, Marker, Popup } from "react-leaflet";
import dynamic from "next/dynamic";
import L from "leaflet";
import "leaflet/dist/leaflet.css";

type MapLocation = {
  player_id: string;
  latitude: number;
  longitude: number;
  accuracy: number | null;
};

type MapPlayer = {
  id: string;
  display_name: string;
};

type GameMapProps = {
  locations: MapLocation[];
  players: MapPlayer[];
};

const playerIcon = new L.Icon({
  iconUrl: "https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png",
  iconRetinaUrl:
    "https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png",
  shadowUrl: "https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png",
  iconSize: [25, 41],
  iconAnchor: [12, 41],
  popupAnchor: [1, -34],
  shadowSize: [41, 41],
});

function GameMap({ locations, players }: GameMapProps) {
  const visibleLocations = locations.filter(
    (location) =>
      Number.isFinite(location.latitude) &&
      Number.isFinite(location.longitude)
  );

  if (visibleLocations.length === 0) {
    return (
      <div className="rounded-2xl border border-slate-800 p-6 text-center text-slate-400">
        Nog geen exacte GPS-locatie ontvangen.
      </div>
    );
  }

  const center: [number, number] = [
    visibleLocations[0].latitude,
    visibleLocations[0].longitude,
  ];

  return (
    <div className="overflow-hidden rounded-2xl border border-slate-800">
      <MapContainer
        center={center}
        zoom={15}
        scrollWheelZoom
        style={{ height: "420px", width: "100%" }}
      >
        <TileLayer
          attribution='&copy; OpenStreetMap contributors'
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />

        {visibleLocations.map((location) => {
          const player = players.find(
            (item) => item.id === location.player_id
          );

          return (
            <Marker
              key={location.player_id}
              position={[location.latitude, location.longitude]}
              icon={playerIcon}
            >
              <Popup>
                <strong>{player?.display_name ?? "Speler"}</strong>
                {location.accuracy !== null && (
                  <div>Nauwkeurigheid: ±{Math.round(location.accuracy)} m</div>
                )}
              </Popup>
            </Marker>
          );
        })}
      </MapContainer>
    </div>
  );
}

export default dynamic(() => Promise.resolve(GameMap), {
  ssr: false,
});

"use client";

import { FormEvent, useState } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "../../lib/supabase";

export default function JoinGame() {
  const router = useRouter();
  const supabase = createClient();

  const [code, setCode] = useState("");
  const [name, setName] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");

    const cleanCode = code.trim().toUpperCase();
    const cleanName = name.trim();

    if (cleanCode.length !== 6) {
      setError("Een spelcode bestaat uit 6 tekens.");
      return;
    }

    if (!cleanName) {
      setError("Vul je naam in.");
      return;
    }

    setLoading(true);

    const { data, error: supabaseError } = await supabase.rpc(
      "join_game",
      {
        p_game_code: cleanCode,
        p_display_name: cleanName,
      }
    );

    setLoading(false);

    if (supabaseError) {
      setError(supabaseError.message);
      return;
    }

    if (!data?.game_id || !data?.player_id || !data?.access_token) {
      setError("Je kon niet aan het spel deelnemen.");
      return;
    }

    sessionStorage.setItem(
      `bankzitters-player-${data.game_id}`,
      JSON.stringify({
        playerId: data.player_id,
        accessToken: data.access_token,
      })
    );

    router.push(
      `/game?code=${encodeURIComponent(data.game_code)}&game=${encodeURIComponent(data.game_id)}`
    );
  }

  return (
    <main className="min-h-screen bg-slate-950 text-white p-6">
      <div className="mx-auto max-w-md">
        <a
          href="/"
          className="inline-block mb-8 text-slate-400 hover:text-white"
        >
          ? Terug
        </a>

        <h1 className="text-3xl font-bold mb-2">
          Meedoen met een spel
        </h1>

        <p className="text-slate-300 mb-8">
          Vul de spelcode en je naam in.
        </p>

        <form onSubmit={handleSubmit} className="space-y-5">
          <div>
            <label className="block mb-2 font-medium">
              Spelcode
            </label>

            <input
              type="text"
              value={code}
              onChange={(event) =>
                setCode(event.target.value.toUpperCase())
              }
              placeholder="Bijvoorbeeld ABC123"
              maxLength={6}
              className="w-full rounded-xl bg-slate-800 border border-slate-700 px-4 py-3 text-center text-2xl font-bold tracking-[0.3em] uppercase outline-none focus:border-blue-500"
            />
          </div>

          <div>
            <label className="block mb-2 font-medium">
              Jouw naam
            </label>

            <input
              type="text"
              value={name}
              onChange={(event) => setName(event.target.value)}
              placeholder="Bijvoorbeeld Sem"
              maxLength={30}
              className="w-full rounded-xl bg-slate-800 border border-slate-700 px-4 py-3 outline-none focus:border-blue-500"
            />
          </div>

          {error && (
            <div className="rounded-xl border border-red-500/40 bg-red-500/10 p-4 text-red-300">
              {error}
            </div>
          )}

          <button
            type="submit"
            disabled={loading}
            className="w-full rounded-xl bg-blue-600 px-6 py-4 text-lg font-semibold hover:bg-blue-500 disabled:opacity-50"
          >
            {loading ? "Bezig..." : "Meedoen"}
          </button>
        </form>
      </div>
    </main>
  );
}

-- SPEURTOCHT APP - DATABASE STRUCTUUR
-- Tijdelijke speldata, ontworpen voor €0 gebruik
-- =========================================================

create table if not exists games (
  id uuid primary key default gen_random_uuid(),
  game_code text unique not null,
  name text not null,
  status text not null default 'waiting',
  start_time timestamptz not null,
  end_time timestamptz not null,
  current_seeker_id uuid,
  created_at timestamptz not null default now()
);

create table if not exists players (
  id uuid primary key default gen_random_uuid(),
  game_id uuid not null references games(id) on delete cascade,
  user_id uuid,
  display_name text not null,
  role text not null default 'player',
  location_sharing boolean not null default false,
  is_found boolean not null default false,
  joined_at timestamptz not null default now()
);

create table if not exists missions (
  id uuid primary key default gen_random_uuid(),
  game_id uuid not null references games(id) on delete cascade,
  title text not null,
  description text,
  start_time timestamptz,
  order_number integer not null default 1,
  active boolean not null default false
);

create table if not exists locations (
  player_id uuid primary key references players(id) on delete cascade,
  latitude double precision not null,
  longitude double precision not null,
  accuracy double precision,
  updated_at timestamptz not null default now()
);

create table if not exists game_events (
  id uuid primary key default gen_random_uuid(),
  game_id uuid not null references games(id) on delete cascade,
  type text not null,
  actor_player_id uuid,
  target_player_id uuid,
  metadata jsonb,
  created_at timestamptz not null default now()
);

-- Indexen voor snelle live-game queries
create index if not exists players_game_id_idx
  on players(game_id);

create index if not exists missions_game_id_idx
  on missions(game_id);

create index if not exists locations_updated_at_idx
  on locations(updated_at);

create index if not exists game_events_game_id_idx
  on game_events(game_id);

-- Locaties worden automatisch verwijderd zodra een speler
-- of een volledig spel wordt verwijderd.
--
-- De locaties-tabel bewaart per speler slechts één actuele
-- locatie. Er ontstaat dus geen historische GPS-route.

-- Speldata ouder dan 24 uur kan later automatisch worden
-- opgeschoond via een scheduled databasefunctie.
